148 template <typename T> struct Triangle2D {
149 Point2D<T> p1;
150 Point2D<T> p2;
151 Point2D<T> p3;
152
153 constexpr Triangle2D() noexcept = default;
154 constexpr Triangle2D(const Point2D<T>& p1_, const Point2D<T>& p2_,
155 const Point2D<T>& p3_) noexcept
156 : p1(p1_), p2(p2_), p3(p3_) {}
157
160 constexpr std::tuple<T, T, T> quadrances() const noexcept {
161 T q1 = (p2.x - p3.x) * (p2.x - p3.x) + (p2.y - p3.y) * (p2.y - p3.y);
162 T q2 = (p1.x - p3.x) * (p1.x - p3.x) + (p1.y - p3.y) * (p1.y - p3.y);
163 T q3 = (p1.x - p2.x) * (p1.x - p2.x) + (p1.y - p2.y) * (p1.y - p2.y);
164 return std::make_tuple(q1, q2, q3);
165 }
166
168 constexpr T area() const noexcept {
169 auto [q1, q2, q3] = quadrances();
170 T temp = q1 + q2 - q3;
171 return T(4) * q1 * q2 - temp * temp;
172 }
173
175 constexpr T twist() const noexcept {
176 return (p2.x - p1.x) * (p3.y - p1.y) - (p2.y - p1.y) * (p3.x - p1.x);
177 }
178
180 constexpr bool is_degenerate() const noexcept { return T(0) == twist(); }
181 };
