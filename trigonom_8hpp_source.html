170 template <typename T> constexpr T archimedes(const T& q_1, const T& q_2, const T& q_3) {
171 validate_no_overflow_multiply(q_1, q_2, "q_1 in archimedes");

530 template <typename T> constexpr Triangle<T> create_triangle(const std::vector<T>& a,
531 const std::vector<T>& b,
532 const std::vector<T>& c) {
533 validate_vector_size(a, 2);
534 validate_vector_size(b, 2);
535 validate_vector_size(c, 2);
536
537 Triangle<T> tri;
538 tri.a = a;
539 tri.b = b;
540 tri.c = c;
541
542 // Calculate side vectors
543 std::vector<T> ab = {b[0] - a[0], b[1] - a[1]};
544 std::vector<T> bc = {c[0] - b[0], c[1] - b[1]};
545 std::vector<T> ca = {a[0] - c[0], a[1] - c[1]};
546
547 // Calculate quadrances
548 tri.q_ab = quad(ab);
549 tri.q_bc = quad(bc);
550 tri.q_ca = quad(ca);
551
552 // Calculate quadrea
553 tri.quadrea = archimedes(tri.q_ab, tri.q_bc, tri.q_ca);
554
555 // Validate triangle is non-degenerate
556 validate_non_zero(tri.quadrea, "quadrea (collinear points)");
557
558 // Calculate spreads
559 tri.spread_a = spread_law(tri.q_ca, tri.q_ab, tri.q_bc);
560 tri.spread_b = spread_law(tri.q_ab, tri.q_bc, tri.q_ca);
561 tri.spread_c = spread_law(tri.q_bc, tri.q_ca, tri.q_ab);
562
563 return tri;
564 }

576 template <typename T>
577 constexpr Triangle<T> solve_triangle_sas(const T& q_1, const T& q_2, const T& spread) {
578 validate_non_negative(q_1, "q_1");
579 validate_non_negative(q_2, "q_2");
580 validate_non_negative(spread, "spread");
581 validate_non_zero(q_1, "q_1");
582 validate_non_zero(q_2, "q_2");
583 validate_non_negative(spread, "spread");
584
585 // Calculate third quadrance using triple quad formula
586 // Note: triple_quad_formula returns the squared third quadrance
587 T q_3_squared = triple_quad_formula(q_1, q_2, spread);
588
589 // Take square root to get the actual third quadrance
590 if constexpr (std::is_floating_point_v<T>) {
591 if (q_3_squared < 0) {
592 throw TrigonomError("Invalid triangle: third quadrance squared is negative");
593 }
594 }
595
596 T q_3 = std::sqrt(q_3_squared);
597
598 // Verify triangle is valid (positive quadrea)
599 T q = archimedes(q_1, q_2, q_3);
600 if (q < T(0)) {
601 throw TrigonomError("Invalid triangle: quadrea is negative for SAS case");
602 }
603
604 // Calculate other spreads
605 T spread_2 = spread_law(q_2, q_3, q_1);
606 T spread_3 = spread_law(q_3, q_1, q_2);
607
608 // Create triangle with vertex at origin
609 Triangle<T> tri;
610 tri.a = {T(0), T(0)};
611 tri.b = {q_1, T(0)}; // Side AB along x-axis
612 tri.c = {q_1 * (1 - spread), std::sqrt(q_1 * q_2 * spread)}; // Point C
613
614 tri.q_ab = q_1;
615 tri.q_bc = q_2;
616 tri.q_ca = q_3;
617 tri.quadrea = q;
618 tri.spread_a = spread_3;
619 tri.spread_b = spread_2;
620 tri.spread_c = spread;
621
622 return tri;
623 }
